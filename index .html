// 1. هيكلة بيانات الطالب (Class & Constructor) class Student { constructor(id, name, grades, attendance) { this.id = id; this.name = name; this.grades = grades; // مصفوفة تحتوي على درجات 3 مواد this.attendance = attendance; // نسبة الحضور } // دالة حساب المتوسط باستخدام reduce calculateAverage() { const sum = this.grades.reduce((acc, curr) => acc + curr, 0); return sum / this.grades.length; } // دالة التقدير بناءً على المتوسط getGradeLetter(average) { if (average >= 90) return "A"; else if (average >= 80) return "B"; else if (average >= 70) return "C"; else if (average >= 60) return "D"; else if (average >= 50) return "E"; else return "F"; } // دالة النتيجة النهائية (ناجح / يحتاج مراجعة) getFinalResult(average) { if (average >= 50 && this.attendance >= 75) { return "ناجح"; } else { return "يحتاج مراجعة"; } } } // 3. نظام تشغيل الجامعة (Objects Array) - إنشاء 4 طلاب (مع وجود متعثرين) const studentsList = [ new Student(1, "أحمد محمد", [85, 90, 88], 85), new Student(2, "سارة علي", [45, 48, 42], 80), // متعثر (متوسطه بين 40 و 49) new Student(3, "محمود خالد", [60, 65, 70], 70), new Student(4, "فاطمة عمر", [44, 46, 43], 90) // متعثر (متوسطه بين 40 و 49) ]; // 5. المخرجات: تقرير لكل طالب في المصفوفة الرئيسية console.log("--- تقرير أداء الطلاب العام ---"); studentsList.forEach(student => { const avg = student.calculateAverage(); const letter = student.getGradeLetter(avg); const result = student.getFinalResult(avg); console.log(`الاسم: ${student.name} | المتوسط: ${avg.toFixed(2)} | التقدير: ${letter} | النتيجة: ${result}`); }); // 4. لجنة الرأفة (Array Methods) // الاستخراج: الطلاب الذين تتراوح متوسطاتهم بين 40 و 49 const strugglingStudents = studentsList.filter(student => { const avg = student.calculateAverage(); return avg >= 40 && avg < 50; }); // المعالجة: استخدام map لإضافة 5 درجات رأفة لمتوسطاتهم const adjustedStrugglingStudents = strugglingStudents.map(student => { const oldAvg = student.calculateAverage(); const newAvg = oldAvg + 5; // إضافة 5 درجات رأفة return { name: student.name, oldAverage: oldAvg, newAverage: newAvg, newGradeLetter: student.getGradeLetter(newAvg) }; }); console.log("\n--- مصفوفة الطلاب المتعثرين بعد تطبيق الرأفة ---"); console.table(adjustedStrugglingStudents); // التحليل: استخدام reduce لجمع المتوسطات الجديدة بعد التعديل const totalNewAverages = adjustedStrugglingStudents.reduce((acc, curr) => acc + curr.newAverage, 0); // متوسط الرأفة: مع درع حماية لمنع القسمة على صفر const count = adjustedStrugglingStudents.length; const overallGraceAverage = count > 0 ? totalNewAverages / count : 0; console.log(`\nمتوسط درجات الطلاب المتعثرين بعد التعديل (الرأفة): ${overallGraceAverage.toFixed(2)}`);
